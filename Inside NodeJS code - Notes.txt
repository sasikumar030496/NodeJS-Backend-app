Swagger is used for API documentation and can also be used alternative to POSTMAN

1). Install below packages inside Node application
		npm i swagger-jsdoc@6.0.0
		npm i swagger-ui-express
2). Write below code inside server.js (NodeJS backend app)
//Swagger related code

const swaggerJSDoc = require("swagger-jsdoc");
const swaggerUI = require("swagger-ui-express");

const options = {
  definition: {
    openapi: "3.0.0",
    info: {
      title: "Swagger API documentation for NodeJS and MongoDB project",
      version: "1.0.0",
    },
    servers: [
      {
        url: "http://localhost:3000",
      },
    ],
  },
  apis: [
    "./src/Routes/auth.routes.js",
    "./src/Routes/product.routes.js",
    "./src/Routes/booking.routes.js",
    "./src/Routes/movie.routes.js",
    "./src/Routes/payment.routes.js",
    "./src/Routes/theatre.routes.js",
  ],
};
const swaggerSpec = swaggerJSDoc(options);

const app = express(); // creating app
app.use("/api-docs", swaggerUI.serve, swaggerUI.setup(swaggerSpec));

3). Write below code in API file(product.routes.js)

// Swagger code for creating schema for products(components/schema is mandatory)
/**
 * @swagger
 *  components:
 *      schema:
 *          Products:
 *              type: object
 *              properties:
 *                  name:
 *                      type: string
 *                  description:
 *                      type: string
 *                  category:
 *                      type: string
 *                  price:
 *                      type: integer
 */

// Swagger code for get all products
/**
 * @swagger
 *  /ecom/api/v1/products:
 *  get:
 *      summary: Fetches all products
 *      description: Returns all products  with all details
 *      responses:
 *          200:
 *              description: To fetch product details
 *              content:
 *                  application/json:
 *                      schema:
 *                          type: array
 *                          items:
 *                              $ref: '#components/schema/Products'
 *
 */

// Swagger code for fetch single product
/**
 * @swagger
 *  /ecom/api/v1/products/{id}:
 *  get:
 *      summary: Fetches product with specific ID
 *      description: Returns product  with specific ID
 *      parameters:
 *          - in: path
 *            name: id
 *            required: true
 *            description: ID required
 *            schema:
 *               type: string
 *      responses:
 *          200:
 *              description: To fetch single producct
 *              content:
 *                  application/json:
 *                      schema:
 *                          type: array
 *                          items:
 *                              $ref: '#components/schema/Products'
 *
 */

// Swagger code to create a product
/**
 * @swagger
 *  /ecom/api/v1/products:
 *      post:
 *          summary: Create a product
 *          description: Creates a new product
 *          requestBody:
 *              required: true
 *              content:
 *                  application/json:
 *                      schema:
 *                          $ref: '#components/schema/Products'
 *          responses:
 *              200:
 *                  description: Product created
 */

// Swagger code to update a product
/**
 * @swagger
 *  /ecom/api/v1/products/{id}:
 *  put:
 *      summary: Update product with specific ID
 *      description: Returns updated product  with specific ID
 *      requestBody:
 *              required: true
 *              content:
 *                  application/json:
 *                      schema:
 *                          $ref: '#components/schema/Products'
 *      parameters:
 *          - in: path
 *            name: id
 *            required: true
 *            description: ID required
 *            schema:
 *               type: string
 *      responses:
 *          200:
 *              description: To fetch single producct
 *              content:
 *                  application/json:
 *                      schema:
 *                          type: array
 *                          items:
 *                              $ref: '#components/schema/Products'
 *
 */

// Swagger code to delete a product
/**
 * @swagger
 *  /ecom/api/v1/products/{id}:
 *  delete:
 *      summary: Deletes product with specific ID
 *      description: Deletes product  with specific ID
 *      parameters:
 *          - in: path
 *            name: id
 *            required: true
 *            description: ID required
 *            schema:
 *               type: string
 *      responses:
 *          200:
 *              description: Product Deleted
 *
 */


Important Notes : 
	==> You can create multiple servers and apis in options
	==> Give tab space after each line for clear code reading
	==> @swagger is mandatory (Syntax: /** code */)
	==> You can write multiple responses like success and failure as well
	==> If response is in array give type as array and for items in array you need to specify type and its properties
	==> You can create seperate components as above and can use them using $ref: '#components/schema/Products'
